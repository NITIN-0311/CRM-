class QTextEditLogger(logging.Handler):
    def __init__(self, text_widget):
        super().__init__()
        self.text_widget = text_widget
        self.mutex = QMutex()

    def emit(self, record):
        msg = self.format(record)
        timestamp = datetime.now().strftime("%H:%M:%S")
        formatted_msg = f"[{timestamp}] {msg}"
        
        self.mutex.lock()
        try:
            self.text_widget.append(formatted_msg)
            # Auto-scroll to bottom
            cursor = self.text_widget.textCursor()
            cursor.movePosition(cursor.End)
            self.text_widget.setTextCursor(cursor)
        finally:
            self.mutex.unlock()

# -----------------------------
class WindowsAlertSystem:
    def __init__(self):
        self.vibration_enabled = True
        self.sound_alerts_enabled = True
        self.last_vibration_time = 0
        self.last_sound_time = 0
        self.vibration_cooldown = 0.3
        self.sound_cooldown = 0.2
        
        self.alert_patterns = {
            'critical': {'beep_freq': 1000, 'beep_duration': 800, 'vibration_strength': 65535},
            'warning': {'beep_freq': 800, 'beep_duration': 400, 'vibration_strength': 45000},
            'info': {'beep_freq': 600, 'beep_duration': 200, 'vibration_strength': 25000},
            'obstacle_close': {'beep_freq': 1200, 'beep_duration': 300, 'vibration_strength': 55000},
            'path_clear': {'beep_freq': 400, 'beep_duration': 150, 'vibration_strength': 15000}
        }
        
        self._init_vibration_system()

    def _init_vibration_system(self):
        try:
            self.xinput = ctypes.windll.xinput1_4
            
            class XINPUT_VIBRATION(ctypes.Structure):
                _fields_ = [("wLeftMotorSpeed", wintypes.WORD),
                           ("wRightMotorSpeed", wintypes.WORD)]
            
            self.XINPUT_VIBRATION = XINPUT_VIBRATION
            
            self.controllers_available = []
            for i in range(4):
                vibration = self.XINPUT_VIBRATION(0, 0)
                result = self.xinput.XInputSetState(i, ctypes.byref(vibration))
                if result == 0:
                    self.controllers_available.append(i)
                    
        except Exception as e:
            logging.warning(f"Vibration system initialization failed: {e}")
            self.vibration_enabled = False

    def vibrate_controller(self, strength=30000, duration=0.3):
        if not self.vibration_enabled or not hasattr(self, 'controllers_available'):
            return
        
        current_time = time.time()
        if current_time - self.last_vibration_time < self.vibration_cooldown:
            return
        
        self.last_vibration_time = current_time
        
        try:
            for controller_id in self.controllers_available:
                vibration = self.XINPUT_VIBRATION(strength, strength)
                self.xinput.XInputSetState(controller_id, ctypes.byref(vibration))
            
            def stop_vibration():
                time.sleep(duration)
                try:
                    for controller_id in self.controllers_available:
                        stop_vib = self.XINPUT_VIBRATION(0, 0)
                        self.xinput.XInputSetState(controller_id, ctypes.byref(stop_vib))
                except Exception:
                    pass
            
            threading.Thread(target=stop_vibration, daemon=True).start()
            
        except Exception as e:
            logging.error(f"Controller vibration error: {e}")

    def play_beep(self, frequency, duration):
        current_time = time.time()
        if current_time - self.last_sound_time < self.sound_cooldown:
            return
        
        self.last_sound_time = current_time
        
        try:
            threading.Thread(target=winsound.Beep, args=(frequency, duration), daemon=True).start()
        except Exception:
            pass

    def critical_alert(self, message="Critical obstacle detected!"):
        pattern = self.alert_patterns['critical']
        
        def critical_beep_sequence():
            for i in range(3):
                try:
                    winsound.Beep(pattern['beep_freq'], 200)
                    time.sleep(0.1)
                except Exception:
                    break
        
        threading.Thread(target=critical_beep_sequence, daemon=True).start()
        self.vibrate_controller(pattern['vibration_strength'], 0.8)
        logging.critical(f"CRITICAL ALERT: {message}")

    def warning_alert(self, message="Obstacle detected"):
        pattern = self.alert_patterns['warning']
        self.play_beep(pattern['beep_freq'], pattern['beep_duration'])
        self.vibrate_controller(pattern['vibration_strength'], 0.4)
        logging.warning(f"WARNING: {message}")

    def info_alert(self, message="Information"):
        pattern = self.alert_patterns['info']
        self.play_beep(pattern['beep_freq'], pattern['beep_duration'])
        self.vibrate_controller(pattern['vibration_strength'], 0.2)
        logging.info(f"INFO: {message}")

    def obstacle_alert(self, obstacle_type, distance, direction):
        message = f"{obstacle_type} {distance} {direction}"
        
        if distance in ['very close']:
            self.critical_alert(message)
        elif distance in ['close']:
            self.warning_alert(message)
        else:
            self.info_alert(message)

    def path_clear_alert(self):
        pattern = self.alert_patterns['path_clear']
        
        def clear_path_tone():
            try:
                for freq in [300, 400, 500]:
                    winsound.Beep(freq, 100)
                    time.sleep(0.05)
            except Exception:
                pass
        
        threading.Thread(target=clear_path_tone, daemon=True).start()
        self.vibrate_controller(pattern['vibration_strength'], 0.1)

    def startup_alert(self):
        def startup_sequence():
            try:
                for freq in [400, 500, 600, 800]:
                    winsound.Beep(freq, 150)
                    time.sleep(0.1)
            except Exception:
                pass
        
        threading.Thread(target=startup_sequence, daemon=True).start()
        self.vibrate_controller(20000, 0.5)

class SafeTTS(QThread):
    def __init__(self):
        super().__init__()
        self.engine = None
        self.tts_queue = queue.Queue()
        self.speaking = False
        self.initialized = False
        self.MIN_PAUSE = 1.5
        self.MAX_QUEUE_SIZE = 3
        self._running = True
        
        self._init_engine()
        self.start()

    def _init_engine(self):
        try:
            self.engine = pyttsx3.init()
            voices = self.engine.getProperty('voices')
            
            if voices:
                for voice in voices:
                    if 'female' in voice.name.lower() or 'zira' in voice.name.lower():
                        self.engine.setProperty('voice', voice.id)
                        break
            
            self.engine.setProperty('rate', 150)
            self.engine.setProperty('volume', 0.95)
            self.initialized = True
            logging.info("TTS initialized successfully")
            
        except Exception as e:
            logging.error(f"TTS initialization failed: {e}")
            self.initialized = False

    def run(self):
        while self._running:
            try:
                msg = self.tts_queue.get(timeout=1)
                if msg == "__EXIT__":
                    break
                
                if self.initialized and self.engine:
                    self.speaking = True
                    self.engine.say(msg)
                    self.engine.runAndWait()
                    self.speaking = False
                    logging.info(f"Spoke: {msg}")
                
                time.sleep(self.MIN_PAUSE)
                self.tts_queue.task_done()
                
            except queue.Empty:
                continue
            except Exception as e:
                logging.error(f"TTS worker error: {e}")
                self.speaking = False

    def speak(self, msg, priority=False):
        if not msg or not self._running:
            return
            
        try:
            if priority:
                while not self.tts_queue.empty():
                    try:
                        self.tts_queue.get_nowait()
                    except queue.Empty:
                        break
                self.tts_queue.put(msg)
                logging.info(f"Priority message queued: {msg}")
            
            elif self.tts_queue.qsize() < self.MAX_QUEUE_SIZE:
                self.tts_queue.put(msg)
            
        except Exception as e:
            logging.error(f"TTS queue error: {e}")

    def is_busy(self):
        return self.speaking or not self.tts_queue.empty()

    def shutdown(self):
        self._running = False
        self.tts_queue.put("__EXIT__")
        self.wait()

class NavigationSystem:
    def __init__(self, alert_system, tts_system):
        self.alert_system = alert_system
        self.tts_system = tts_system
        self.confidence_threshold = 0.4
        self.previous_guidance = ""
        self.guidance_history = deque(maxlen=3)
        self.emergency_objects = {'person', 'car', 'truck', 'bus', 'motorcycle', 'bicycle'}
        self.last_clear_path_time = time.time()
        self.clear_path_interval = 8
        self.last_obstacle_count = 0
        self.last_obstacle_alert_time = {}
        self.obstacle_alert_cooldown = 2.0

        # Initialize TensorFlow Lite model
        self.interpreter = self.load_model_safe()
        if self.interpreter:
            self.input_details = self.interpreter.get_input_details()
            self.output_details = self.interpreter.get_output_details()
        
        # Load labels
        self.labels = self.load_labels_safe()

    def load_model_safe(self):
        try:
            interpreter = tf.lite.Interpreter(model_path="efficientdet_lite2_448_ptq.tflite")
            interpreter.allocate_tensors()
            logging.info("TensorFlow Lite model loaded successfully")
            self.tts_system.speak("AI model loaded successfully")
            self.alert_system.info_alert("AI model ready")
            return interpreter
        except Exception as e:
            logging.error(f"Model loading failed: {e}")
            self.tts_system.speak("Warning: AI model failed to load", priority=True)
            self.alert_system.critical_alert("AI model loading failed")
            return None

    def load_labels_safe(self):
        try:
            with open("labels.txt") as f:
                labels = [line.strip() for line in f.readlines()]
            logging.info(f"Loaded {len(labels)} object labels")
            return labels
        except Exception as e:
            logging.error(f"Labels loading failed: {e}")
            self.tts_system.speak("Warning: Object labels not loaded", priority=True)
            self.alert_system.warning_alert("Labels file missing")
            return []

    def get_distance_category(self, box_height, frame_height):
        ratio = frame_height / box_height
        if ratio < 3:
            return "very close", 3
        elif ratio < 6:
            return "close", 2
        elif ratio < 10:
            return "nearby", 1
        else:
            return "far", 0

    def is_critical_obstacle(self, label, distance_priority):
        label_lower = label.lower()
        is_emergency = any(obj in label_lower for obj in self.emergency_objects)
        return is_emergency and distance_priority >= 2

    def get_direction_from_position(self, x_center, width):
        if x_center < width/3:
            return "left"
        elif x_center < 2*width/3:
            return "center"
        else:
            return "right"

    def should_alert_obstacle(self, obstacle_type, distance):
        current_time = time.time()
        key = f"{obstacle_type}_{distance}"
        
        if key in self.last_obstacle_alert_time:
            if current_time - self.last_obstacle_alert_time[key] < self.obstacle_alert_cooldown:
                return False
        
        self.last_obstacle_alert_time[key] = current_time
        return True

    def analyze_obstacles(self, frame):
        if not self.interpreter:
            return {"left": True, "center": True, "right": True}, [], [], frame

        h, w, _ = frame.shape
        input_size = self.input_details[0]['shape'][1]
        
        img = Image.fromarray(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
        img_resized = img.resize((input_size, input_size))
        input_data = np.expand_dims(np.array(img_resized), axis=0).astype(np.uint8)
        
        self.interpreter.set_tensor(self.input_details[0]['index'], input_data)
        self.interpreter.invoke()
        
        boxes = self.interpreter.get_tensor(self.output_details[0]['index'])[0]
        classes = self.interpreter.get_tensor(self.output_details[1]['index'])[0]
        scores = self.interpreter.get_tensor(self.output_details[2]['index'])[0]
        
        lane_status = {"left": True, "center": True, "right": True}
        critical_obstacles = []
        all_obstacles = []
        
        for i in range(len(scores)):
            if scores[i] > self.confidence_threshold:
                if int(classes[i]) >= len(self.labels):
                    continue
                    
                label = self.labels[int(classes[i])]
                ymin, xmin, ymax, xmax = boxes[i]
                
                startX, startY = int(xmin*w), int(ymin*h)
                endX, endY = int(xmax*w), int(ymax*h)
                box_height = endY - startY
                x_center = (startX + endX) / 2
                
                distance_text, distance_priority = self.get_distance_category(box_height, h)
                direction = self.get_direction_from_position(x_center, w)
                
                if distance_priority > 0:
                    obstacle_info = {
                        'label': label,
                        'distance': distance_text,
                        'priority': distance_priority,
                        'confidence': scores[i],
                        'position': (startX, startY, endX, endY),
                        'direction': direction
                    }
                    
                    all_obstacles.append(obstacle_info)
                    
                    if self.should_alert_obstacle(label, distance_text):
                        self.alert_system.obstacle_alert(label, distance_text, direction)
                    
                    if self.is_critical_obstacle(label, distance_priority):
                        critical_obstacles.append(obstacle_info)
                    
                    if x_center < w/3:
                        lane_status["left"] = False
                    elif x_center < 2*w/3:
                        lane_status["center"] = False
                    else:
                        lane_status["right"] = False
                    
                    color = (0, 0, 255) if distance_priority >= 2 else (0, 255, 0)
                    thickness = 3 if distance_priority >= 2 else 2
                    
                    cv2.rectangle(frame, (startX, startY), (endX, endY), color, thickness)
                    cv2.putText(frame, f"{label} ({distance_text})", (startX, startY-10),
                                cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 2)
        
        return lane_status, critical_obstacles, all_obstacles, frame

    def generate_guidance(self, lane_status, critical_obstacles, all_obstacles):
        current_time = time.time()
        
        if critical_obstacles:
            critical_obj = critical_obstacles[0]
            guidance = f"Stop! {critical_obj['label']} {critical_obj['distance']} ahead!"
            self.alert_system.critical_alert(guidance)
            self.tts_system.speak(guidance, priority=True)
            self.previous_guidance = guidance
            return guidance
        
        guidance = self._get_movement_guidance(lane_status)
        
        if all_obstacles:
            nearby_objects = [obj for obj in all_obstacles if obj['priority'] >= 1]
            if nearby_objects:
                obstacle_info = self._format_obstacle_info(nearby_objects[:2])
                full_guidance = f"{guidance}. {obstacle_info}"
            else:
                full_guidance = guidance
        else:
            if current_time - self.last_clear_path_time > self.clear_path_interval:
                full_guidance = "Path is clear, continue straight"
                if self.previous_guidance != full_guidance:
                    self.alert_system.path_clear_alert()
                self.last_clear_path_time = current_time
            else:
                full_guidance = guidance
        
        self.guidance_history.append(guidance)
        
        should_speak = (
            full_guidance != self.previous_guidance or
            len(set(self.guidance_history)) == 1
        )
        
        if should_speak and not self.tts_system.is_busy():
            self.tts_system.speak(full_guidance)
            self.previous_guidance = full_guidance
        
        return full_guidance

    def _get_movement_guidance(self, lane_status):
        left_clear = lane_status["left"]
        center_clear = lane_status["center"]
        right_clear = lane_status["right"]
        
        if center_clear:
            if left_clear and right_clear:
                return "Continue straight"
            else:
                return "Go straight, obstacles on sides"
        elif left_clear and right_clear:
            return "Obstacle ahead, move left or right"
        elif left_clear:
            return "Move left"
        elif right_clear:
            return "Move right"
        else:
            return "Stop! Obstacles in all directions"

    def _format_obstacle_info(self, obstacles):
        if not obstacles:
            return ""
        
        if len(obstacles) == 1:
            obj = obstacles[0]
            return f"{obj['label']} detected {obj['distance']}"
        else:
            obj_list = [f"{obj['label']}" for obj in obstacles]
            return f"Multiple objects: {', '.join(obj_list)}"

class VideoThread(QThread):
    frame_ready = pyqtSignal(np.ndarray)
    status_update = pyqtSignal(str, int, str)  # guidance, obstacle_count, status

    def __init__(self, navigation_system):
        super().__init__()
        self.navigation_system = navigation_system
        self.running = False
        self.cap = None

    def start_capture(self):
        self.cap = cv2.VideoCapture(0)
        if not self.cap.isOpened():
            for i in range(1, 4):
                self.cap = cv2.VideoCapture(i)
                if self.cap.isOpened():
                    logging.info(f"Using camera index {i}")
                    break
        
        if not self.cap.isOpened():
            logging.error("No camera available")
            return False
        
        self.cap.set(cv2.CAP_PROP_FRAME_WIDTH, 640)
        self.cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 480)
        self.cap.set(cv2.CAP_PROP_FPS, 15)
        
        logging.info("Camera initialized successfully")
        return True

    def run(self):
        self.running = True
        
        while self.running:
            if self.cap and self.cap.isOpened():
                ret, frame = self.cap.read()
                if ret:
                    try:
                        lane_status, critical_obstacles, all_obstacles, processed_frame = \
                            self.navigation_system.analyze_obstacles(frame)
                        guidance = self.navigation_system.generate_guidance(
                            lane_status, critical_obstacles, all_obstacles)
                        
                        self.frame_ready.emit(processed_frame)
                        self.status_update.emit(guidance, len(all_obstacles), "Running")
                        
                    except Exception as e:
                        logging.error(f"Processing error: {e}")
            
            self.msleep(33)  # ~30 FPS

    def stop(self):
        self.running = False
        if self.cap:
            self.cap.release()
        self.quit()
        self.wait()

struct node *creation(struct node *head, char line[], int st_num) {		//creation of node
    struct node *ptr = (struct node *) malloc(sizeof(struct node));
    ptr->num = st_num;
    strcpy(ptr->data, line);
    ptr->next = NULL;

    if (head == NULL) {
        head = ptr;
    } else {
        struct node *temp = head;
        while (temp->next != NULL) {
            temp = temp->next;
        }
        temp->next = ptr;
    }
    return head;
}

void traverse(struct node *head) {				//Traversal of Node
    struct node *temp = head;
    while(temp != NULL) {
        printf("%d  %s \n", temp->num, temp->data);
        temp = temp->next;
    }
}

int searching(struct node *head,int start, int stop) {
    struct node *temp = head;
    int count = 1;

    // Find the start and stop nodes
    struct node *start_node = NULL;
    struct node *stop_node = NULL;
    while (temp != NULL)
    {
        if (temp->num == start)
            start_node = temp;
        if (temp->num == stop)
            stop_node = temp;
        temp = temp->next;
    }

    // Count the stations between start and stop
    if (start_node != NULL && stop_node != NULL)
    {
        if (start_node->num < stop_node->num)
        {
            temp = start_node->next;
            while (temp != stop_node)
            {
                count++;
                temp = temp->next;
            }
        }
    else
        {
            temp = stop_node->next;
            while (temp != start_node)
            {
                count++;
                temp = temp->next;
            }
        }
    }

    // Print the number of stations
    return count;
}
